-- =====================================================
-- BASE DE DATOS: BIBLIOTECA DIGITAL
-- COMPATIBLE CON MYSQL 5.7
-- =====================================================

-- Eliminar base de datos si existe (CUIDADO: Borra todo)
-- DROP DATABASE IF EXISTS biblioteca_digital;

CREATE DATABASE IF NOT EXISTS biblioteca_digital
    CHARACTER SET utf8mb4
    COLLATE utf8mb4_unicode_ci;

USE biblioteca_digital;

-- =====================================================
-- TABLA: USUARIO
-- =====================================================
DROP TABLE IF EXISTS usuario;
CREATE TABLE usuario (
                         id_usuario INT NOT NULL AUTO_INCREMENT,
                         dni VARCHAR(20) NOT NULL,
                         nombre VARCHAR(100) NOT NULL,
                         email VARCHAR(100) NOT NULL,
                         telefono VARCHAR(20),
                         direccion VARCHAR(200),
                         tipo_usuario ENUM('ADMIN', 'BIBLIOTECARIO', 'LECTOR') NOT NULL,
                         password_hash VARCHAR(255) NOT NULL,
    -- MySQL 5.7 no soporta DEFAULT (CURRENT_DATE), usar CURRENT_TIMESTAMP
                         fecha_registro DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
                         estado ENUM('ACTIVO', 'SUSPENDIDO', 'MOROSO', 'INACTIVO') NOT NULL DEFAULT 'ACTIVO',
                         limite_prestamos INT NOT NULL DEFAULT 3,
                         fecha_ultima_actividad DATETIME,

                         PRIMARY KEY (id_usuario),
                         UNIQUE KEY unique_dni (dni),
                         UNIQUE KEY unique_email (email),
                         INDEX idx_dni (dni),
                         INDEX idx_email (email),
                         INDEX idx_tipo_usuario (tipo_usuario),
                         INDEX idx_estado (estado)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- =====================================================
-- TABLA: CATEGORIA
-- =====================================================
DROP TABLE IF EXISTS categoria;
CREATE TABLE categoria (
                           id_categoria INT NOT NULL AUTO_INCREMENT,
                           nombre VARCHAR(50) NOT NULL,
                           descripcion TEXT,
                           categoria_padre INT DEFAULT NULL,
    -- MySQL 5.7 usa TINYINT(1) en lugar de BOOLEAN
                           activa TINYINT(1) NOT NULL DEFAULT 1,
                           codigo VARCHAR(10) DEFAULT NULL,

                           PRIMARY KEY (id_categoria),
                           UNIQUE KEY unique_nombre (nombre),
                           UNIQUE KEY unique_codigo (codigo),
                           INDEX idx_nombre (nombre),
                           FOREIGN KEY (categoria_padre) REFERENCES categoria(id_categoria)
                               ON DELETE SET NULL
                               ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- =====================================================
-- TABLA: EDITORIAL (NUEVA - Normalización)
-- =====================================================
DROP TABLE IF EXISTS editorial;
CREATE TABLE editorial (
                           id_editorial INT NOT NULL AUTO_INCREMENT,
                           nombre VARCHAR(100) NOT NULL,
                           pais VARCHAR(50),
                           ciudad VARCHAR(50),
                           telefono VARCHAR(20),
                           email VARCHAR(100),
                           sitio_web VARCHAR(200),
                           activa TINYINT(1) NOT NULL DEFAULT 1,

                           PRIMARY KEY (id_editorial),
                           UNIQUE KEY unique_nombre (nombre),
                           INDEX idx_nombre (nombre)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- =====================================================
-- TABLA: AUTOR
-- =====================================================
DROP TABLE IF EXISTS autor;
CREATE TABLE autor (
                       id_autor INT NOT NULL AUTO_INCREMENT,
                       nombre_completo VARCHAR(150) NOT NULL,
                       nacionalidad VARCHAR(50),
                       fecha_nacimiento DATE,
                       fecha_fallecimiento DATE,
                       biografia TEXT,
                       premios TEXT,

                       PRIMARY KEY (id_autor),
                       INDEX idx_nombre (nombre_completo),
                       INDEX idx_nacionalidad (nacionalidad)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- =====================================================
-- TABLA: LIBRO
-- =====================================================
DROP TABLE IF EXISTS libro;
CREATE TABLE libro (
                       id_libro INT NOT NULL AUTO_INCREMENT,
                       isbn VARCHAR(20) NOT NULL,
                       titulo VARCHAR(200) NOT NULL,
    -- Campos para compatibilidad con BD existente
                       autor VARCHAR(150) NOT NULL COMMENT 'Campo legacy - usar libro_autor',
                       editorial VARCHAR(100) DEFAULT NULL COMMENT 'Campo legacy - usar id_editorial',
    -- Campo normalizado (FK a tabla editorial)
                       id_editorial INT NOT NULL,
                       año_publicacion INT NOT NULL,
                       id_categoria INT NOT NULL,
                       num_paginas INT,
                       idioma VARCHAR(30) NOT NULL DEFAULT 'Español',
                       copias_totales INT NOT NULL DEFAULT 1,
                       copias_disponibles INT NOT NULL DEFAULT 1,
                       ubicacion VARCHAR(50) NOT NULL,
                       descripcion TEXT,
                       portada_url VARCHAR(255),
                       fecha_adquisicion DATE NOT NULL,
                       precio DECIMAL(10,2),
                       estado_fisico ENUM('EXCELENTE', 'BUENO', 'REGULAR', 'MALO') NOT NULL DEFAULT 'BUENO',

                       PRIMARY KEY (id_libro),
                       UNIQUE KEY unique_isbn (isbn),
                       INDEX idx_isbn (isbn),
                       INDEX idx_titulo (titulo),
                       INDEX idx_autor (autor),
                       INDEX idx_categoria (id_categoria),
                       INDEX idx_editorial (id_editorial),
                       INDEX idx_disponibilidad (copias_disponibles),
                       FULLTEXT INDEX ft_busqueda (titulo, autor, descripcion),

                       FOREIGN KEY (id_categoria) REFERENCES categoria(id_categoria)
                           ON DELETE RESTRICT
                           ON UPDATE CASCADE,
                       FOREIGN KEY (id_editorial) REFERENCES editorial(id_editorial)
                           ON DELETE RESTRICT
                           ON UPDATE CASCADE

    -- MySQL 5.7 NO soporta CHECK constraints, se usarán TRIGGERS
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- =====================================================
-- TABLA: LIBRO_AUTOR (Relación N:M)
-- =====================================================
DROP TABLE IF EXISTS libro_autor;
CREATE TABLE libro_autor (
                             id_libro INT NOT NULL,
                             id_autor INT NOT NULL,
                             orden INT NOT NULL DEFAULT 1,
                             rol VARCHAR(50) DEFAULT 'Autor',

                             PRIMARY KEY (id_libro, id_autor),

                             FOREIGN KEY (id_libro) REFERENCES libro(id_libro)
                                 ON DELETE CASCADE
                                 ON UPDATE CASCADE,
                             FOREIGN KEY (id_autor) REFERENCES autor(id_autor)
                                 ON DELETE CASCADE
                                 ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- =====================================================
-- TABLA: PRESTAMO
-- =====================================================
DROP TABLE IF EXISTS prestamo;
CREATE TABLE prestamo (
                          id_prestamo INT NOT NULL AUTO_INCREMENT,
                          id_libro INT NOT NULL,
                          id_usuario INT NOT NULL,
                          id_bibliotecario INT NOT NULL,
                          fecha_prestamo DATE NOT NULL,
                          fecha_devolucion_esperada DATE NOT NULL,
                          fecha_devolucion_real DATE,
                          estado ENUM('ACTIVO', 'DEVUELTO', 'VENCIDO', 'RENOVADO') NOT NULL DEFAULT 'ACTIVO',
                          renovaciones_realizadas INT NOT NULL DEFAULT 0,
                          max_renovaciones INT NOT NULL DEFAULT 2,
                          observaciones TEXT,

                          PRIMARY KEY (id_prestamo),
                          INDEX idx_libro (id_libro),
                          INDEX idx_usuario (id_usuario),
                          INDEX idx_bibliotecario (id_bibliotecario),
                          INDEX idx_estado (estado),
                          INDEX idx_fecha_prestamo (fecha_prestamo),
                          INDEX idx_fecha_devolucion (fecha_devolucion_esperada),

                          FOREIGN KEY (id_libro) REFERENCES libro(id_libro)
                              ON DELETE RESTRICT
                              ON UPDATE CASCADE,
                          FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario)
                              ON DELETE RESTRICT
                              ON UPDATE CASCADE,
                          FOREIGN KEY (id_bibliotecario) REFERENCES usuario(id_usuario)
                              ON DELETE RESTRICT
                              ON UPDATE CASCADE

    -- CHECK constraints simulados con TRIGGERS
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- =====================================================
-- TABLA: MULTA
-- =====================================================
DROP TABLE IF EXISTS multa;
CREATE TABLE multa (
                       id_multa INT NOT NULL AUTO_INCREMENT,
                       id_prestamo INT NOT NULL,
                       monto DECIMAL(10,2) NOT NULL,
                       monto_por_dia DECIMAL(10,2) NOT NULL DEFAULT 10.00,
                       dias_retraso INT NOT NULL,
                       fecha_generacion DATE NOT NULL,
                       fecha_pago DATE,
                       metodo_pago ENUM('EFECTIVO', 'TARJETA', 'TRANSFERENCIA', 'OTRO'),
                       estado ENUM('PENDIENTE', 'PAGADA', 'CONDONADA') NOT NULL DEFAULT 'PENDIENTE',
                       id_usuario_recibe_pago INT,
                       observaciones TEXT,

                       PRIMARY KEY (id_multa),
                       INDEX idx_prestamo (id_prestamo),
                       INDEX idx_estado (estado),
                       INDEX idx_fecha_generacion (fecha_generacion),

                       FOREIGN KEY (id_prestamo) REFERENCES prestamo(id_prestamo)
                           ON DELETE RESTRICT
                           ON UPDATE CASCADE,
                       FOREIGN KEY (id_usuario_recibe_pago) REFERENCES usuario(id_usuario)
                           ON DELETE SET NULL
                           ON UPDATE CASCADE

    -- CHECK constraints simulados con TRIGGERS
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- =====================================================
-- TABLA: RESERVA
-- =====================================================
DROP TABLE IF EXISTS reserva;
CREATE TABLE reserva (
                         id_reserva INT NOT NULL AUTO_INCREMENT,
                         id_libro INT NOT NULL,
                         id_usuario INT NOT NULL,
                         fecha_reserva DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
                         fecha_vencimiento DATE NOT NULL,
                         estado ENUM('PENDIENTE', 'DISPONIBLE', 'RECOGIDA', 'CANCELADA', 'VENCIDA') NOT NULL DEFAULT 'PENDIENTE',
                         fecha_notificacion DATETIME,
                         prioridad INT NOT NULL DEFAULT 0,
                         observaciones TEXT,

                         PRIMARY KEY (id_reserva),
                         INDEX idx_libro (id_libro),
                         INDEX idx_usuario (id_usuario),
                         INDEX idx_estado (estado),
                         INDEX idx_prioridad (prioridad),
                         INDEX idx_fecha_reserva (fecha_reserva),

                         FOREIGN KEY (id_libro) REFERENCES libro(id_libro)
                             ON DELETE CASCADE
                             ON UPDATE CASCADE,
                         FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario)
                             ON DELETE CASCADE
                             ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- =====================================================
-- TABLA: AUDITORIA (Opcional)
-- =====================================================
DROP TABLE IF EXISTS auditoria;
CREATE TABLE auditoria (
                           id_auditoria INT NOT NULL AUTO_INCREMENT,
                           id_usuario INT NOT NULL,
                           accion VARCHAR(100) NOT NULL,
                           tabla_afectada VARCHAR(50) NOT NULL,
                           id_registro INT NOT NULL,
                           fecha_hora DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
                           ip_address VARCHAR(45),
                           detalles TEXT,

                           PRIMARY KEY (id_auditoria),
                           INDEX idx_usuario (id_usuario),
                           INDEX idx_fecha_hora (fecha_hora),
                           INDEX idx_tabla (tabla_afectada),

                           FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario)
                               ON DELETE RESTRICT
                               ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- =====================================================
-- TABLA: CONFIGURACION (Opcional)
-- =====================================================
DROP TABLE IF EXISTS configuracion;
CREATE TABLE configuracion (
                               id_config INT NOT NULL AUTO_INCREMENT,
                               clave VARCHAR(50) NOT NULL,
                               valor VARCHAR(255) NOT NULL,
                               descripcion TEXT,
                               tipo_dato ENUM('STRING', 'INT', 'DECIMAL', 'BOOLEAN', 'DATE') NOT NULL,

                               PRIMARY KEY (id_config),
                               UNIQUE KEY unique_clave (clave)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- =====================================================
-- ÍNDICES ADICIONALES PARA MEJOR RENDIMIENTO
-- =====================================================
ALTER TABLE libro ADD INDEX idx_año_publicacion (año_publicacion);
ALTER TABLE libro ADD INDEX idx_estado_fisico (estado_fisico);
ALTER TABLE libro ADD INDEX idx_fecha_adquisicion (fecha_adquisicion);
ALTER TABLE usuario ADD INDEX idx_fecha_registro (fecha_registro);
ALTER TABLE prestamo ADD INDEX idx_fecha_real (fecha_devolucion_real);
ALTER TABLE multa ADD INDEX idx_fecha_pago (fecha_pago);
ALTER TABLE reserva ADD INDEX idx_fecha_vencimiento (fecha_vencimiento);

    BEFORE INSERT ON libro
    FOR EACH ROW
BEGIN
    IF NEW.copias_disponibles > NEW.copias_totales THEN
        SIGNAL SQLSTATE '45000'
            SET MESSAGE_TEXT = 'Error: copias_disponibles no puede ser mayor que copias_totales';
    END IF;

    IF NEW.copias_disponibles < 0 THEN
        SIGNAL SQLSTATE '45000'
            SET MESSAGE_TEXT = 'Error: copias_disponibles no puede ser negativo';
    END IF;

    IF NEW.año_publicacion > YEAR(CURDATE()) THEN
        SIGNAL SQLSTATE '45000'
            SET MESSAGE_TEXT = 'Error: año_publicacion no puede ser futuro';
    END IF;
END$$

-- Vista 1: Libros disponibles para préstamo
CREATE OR REPLACE VIEW v_libros_disponibles AS
SELECT
    l.id_libro, l.isbn, l.titulo, l.autor, e.nombre as editorial,
    c.nombre as categoria, l.copias_disponibles, l.ubicacion,
    l.estado_fisico, l.idioma, l.año_publicacion
FROM libro l
         INNER JOIN editorial e ON l.id_editorial = e.id_editorial
         INNER JOIN categoria c ON l.id_categoria = c.id_categoria
WHERE l.copias_disponibles > 0
  AND c.activa = 1
  AND e.activa = 1
ORDER BY l.titulo;

DELIMITER $$

-- Vista 2: Préstamos activos con información detallada
CREATE OR REPLACE VIEW v_prestamos_activos AS
SELECT
    p.id_prestamo, l.titulo as libro, l.isbn,
    u.nombre as usuario, u.email as email_usuario,
    u_bibliotecario.nombre as bibliotecario,
    p.fecha_prestamo, p.fecha_devolucion_esperada,
    DATEDIFF(CURDATE(), p.fecha_devolucion_esperada) as dias_vencido,
    p.renovaciones_realizadas, p.max_renovaciones
FROM prestamo p
         INNER JOIN libro l ON p.id_libro = l.id_libro
         INNER JOIN usuario u ON p.id_usuario = u.id_usuario
         INNER JOIN usuario u_bibliotecario ON p.id_bibliotecario = u_bibliotecario.id_usuario
WHERE p.estado = 'ACTIVO'
ORDER BY p.fecha_devolucion_esperada ASC;

-- Vista 3: Multas pendientes
CREATE OR REPLACE VIEW v_multas_pendientes AS
SELECT
    m.id_multa, u.dni, u.nombre as usuario, u.email, u.telefono,
    l.titulo as libro, p.fecha_devolucion_esperada,
    m.monto, m.dias_retraso, m.fecha_generacion,
    DATEDIFF(CURDATE(), m.fecha_generacion) as dias_generada
FROM multa m
         INNER JOIN prestamo p ON m.id_prestamo = p.id_prestamo
         INNER JOIN usuario u ON p.id_usuario = u.id_usuario
         INNER JOIN libro l ON p.id_libro = l.id_libro
WHERE m.estado = 'PENDIENTE'
ORDER BY m.fecha_generacion DESC;

-- Vista 4: Estadísticas de uso por usuario
CREATE OR REPLACE VIEW v_estadisticas_usuarios AS
SELECT
    u.id_usuario, u.dni, u.nombre, u.email, u.tipo_usuario,
    u.fecha_registro, u.estado,
    COUNT(DISTINCT p.id_prestamo) as total_prestamos,
    COUNT(DISTINCT CASE WHEN p.estado = 'ACTIVO' THEN p.id_prestamo END) as prestamos_activos,
    COUNT(DISTINCT r.id_reserva) as total_reservas,
    COUNT(DISTINCT m.id_multa) as total_multas,
    COALESCE(SUM(CASE WHEN m.estado = 'PENDIENTE' THEN m.monto END), 0) as multas_pendientes
FROM usuario u
         LEFT JOIN prestamo p ON u.id_usuario = p.id_usuario
         LEFT JOIN reserva r ON u.id_usuario = r.id_usuario
         LEFT JOIN multa m ON p.id_prestamo = m.id_prestamo
GROUP BY u.id_usuario, u.dni, u.nombre, u.email, u.tipo_usuario, u.fecha_registro, u.estado
ORDER BY u.tipo_usuario, u.nombre;

-- Vista 5: Libros más populares (por préstamos)
CREATE OR REPLACE VIEW v_libros_populares AS
SELECT
    l.id_libro, l.isbn, l.titulo, l.autor, e.nombre as editorial,
    c.nombre as categoria,
    COUNT(p.id_prestamo) as veces_prestado,
    l.copias_totales, l.copias_disponibles,
    ROUND((COUNT(p.id_prestamo) / l.copias_totales), 2) as tasa_uso
FROM libro l
         INNER JOIN editorial e ON l.id_editorial = e.id_editorial
         INNER JOIN categoria c ON l.id_categoria = c.id_categoria
         LEFT JOIN prestamo p ON l.id_libro = p.id_libro
GROUP BY l.id_libro, l.isbn, l.titulo, l.autor, e.nombre, c.nombre, l.copias_totales, l.copias_disponibles
ORDER BY veces_prestado DESC
LIMIT 20;

-- Procedimiento 1: Registrar nuevo préstamo
DROP PROCEDURE IF EXISTS sp_registrar_prestamo$$
CREATE PROCEDURE sp_registrar_prestamo(
    IN p_id_libro INT,
    IN p_id_usuario INT,
    IN p_id_bibliotecario INT,
    IN p_dias_prestamo INT,
    OUT p_resultado VARCHAR(200)
)
BEGIN
    DECLARE v_copias_disponibles INT;
    DECLARE v_limite_prestamos INT;
    DECLARE v_prestamos_activos INT;
    DECLARE v_estado_usuario VARCHAR(20);

    -- Verificar estado del usuario
    SELECT estado, limite_prestamos INTO v_estado_usuario, v_limite_prestamos
    FROM usuario WHERE id_usuario = p_id_usuario;

    IF v_estado_usuario != 'ACTIVO' THEN
        SET p_resultado = CONCAT('Error: Usuario no activo. Estado: ', v_estado_usuario);
    ELSE
        -- Verificar límite de préstamos
        SELECT COUNT(*) INTO v_prestamos_activos
        FROM prestamo
        WHERE id_usuario = p_id_usuario AND estado = 'ACTIVO';

        IF v_prestamos_activos >= v_limite_prestamos THEN
            SET p_resultado = CONCAT('Error: Usuario alcanzó el límite de préstamos (', v_limite_prestamos, ')');
        ELSE
            -- Verificar disponibilidad del libro
            SELECT copias_disponibles INTO v_copias_disponibles
            FROM libro WHERE id_libro = p_id_libro;

            IF v_copias_disponibles <= 0 THEN
                SET p_resultado = 'Error: No hay copias disponibles del libro';
            ELSE
                -- Registrar préstamo
                INSERT INTO prestamo (id_libro, id_usuario, id_bibliotecario,
                                      fecha_prestamo, fecha_devolucion_esperada, estado)
                VALUES (p_id_libro, p_id_usuario, p_id_bibliotecario,
                        CURDATE(), DATE_ADD(CURDATE(), INTERVAL p_dias_prestamo DAY), 'ACTIVO');

                SET p_resultado = CONCAT('OK: Préstamo registrado. ID: ', LAST_INSERT_ID());
            END IF;
        END IF;
    END IF;
END$$

-- Procedimiento 2: Registrar devolución
DROP PROCEDURE IF EXISTS sp_registrar_devolucion$$
CREATE PROCEDURE sp_registrar_devolucion(
    IN p_id_prestamo INT,
    IN p_observaciones TEXT,
    OUT p_resultado VARCHAR(200),
    OUT p_dias_retraso INT,
    OUT p_multa_generada DECIMAL(10,2)
)
BEGIN
    DECLARE v_fecha_devolucion_esperada DATE;
    DECLARE v_estado VARCHAR(20);
    DECLARE v_multa_por_dia DECIMAL(10,2);

    -- Obtener datos del préstamo
    SELECT fecha_devolucion_esperada, estado
    INTO v_fecha_devolucion_esperada, v_estado
    FROM prestamo WHERE id_prestamo = p_id_prestamo;

    IF v_estado != 'ACTIVO' THEN
        SET p_resultado = CONCAT('Error: Préstamo no está activo. Estado: ', v_estado);
        SET p_dias_retraso = 0;
        SET p_multa_generada = 0;
    ELSE
        -- Calcular días de retraso
        SET p_dias_retraso = DATEDIFF(CURDATE(), v_fecha_devolucion_esperada);
        IF p_dias_retraso < 0 THEN
            SET p_dias_retraso = 0;
        END IF;

        -- Actualizar préstamo
        UPDATE prestamo
        SET estado = 'DEVUELTO',
            fecha_devolucion_real = CURDATE(),
            observaciones = CONCAT(IFNULL(observaciones, ''), ' ', p_observaciones)
        WHERE id_prestamo = p_id_prestamo;

        -- Generar multa si hay retraso
        IF p_dias_retraso > 0 THEN
            SELECT CAST(valor AS DECIMAL(10,2)) INTO v_multa_por_dia
            FROM configuracion WHERE clave = 'multa_por_dia';

            SET p_multa_generada = v_multa_por_dia * p_dias_retraso;

            INSERT INTO multa (id_prestamo, monto, monto_por_dia,
                               dias_retraso, fecha_generacion, estado)
            VALUES (p_id_prestamo, p_multa_generada, v_multa_por_dia,
                    p_dias_retraso, CURDATE(), 'PENDIENTE');

            SET p_resultado = CONCAT('OK: Devolución registrada con multa de $',
                                     p_multa_generada, ' (', p_dias_retraso, ' días)');
        ELSE
            SET p_multa_generada = 0;
            SET p_resultado = 'OK: Devolución registrada a tiempo';
        END IF;
    END IF;
END$$

-- Procedimiento 3: Reservar libro
DROP PROCEDURE IF EXISTS sp_reservar_libro$$
CREATE PROCEDURE sp_reservar_libro(
    IN p_id_libro INT,
    IN p_id_usuario INT,
    IN p_dias_vigencia INT,
    OUT p_resultado VARCHAR(200),
    OUT p_id_reserva INT
)
BEGIN
    DECLARE v_copias_disponibles INT;
    DECLARE v_reservas_activas INT;

    -- Verificar disponibilidad
    SELECT copias_disponibles INTO v_copias_disponibles
    FROM libro WHERE id_libro = p_id_libro;

    IF v_copias_disponibles > 0 THEN
        SET p_resultado = 'Error: El libro está disponible para préstamo inmediato';
        SET p_id_reserva = NULL;
    ELSE
        -- Verificar si ya tiene reserva activa para este libro
        SELECT COUNT(*) INTO v_reservas_activas
        FROM reserva
        WHERE id_libro = p_id_libro
          AND id_usuario = p_id_usuario
          AND estado IN ('PENDIENTE', 'DISPONIBLE');

        IF v_reservas_activas > 0 THEN
            SET p_resultado = 'Error: Ya tienes una reserva activa para este libro';
            SET p_id_reserva = NULL;
        ELSE
            -- Crear reserva
            INSERT INTO reserva (id_libro, id_usuario, fecha_reserva,
                                 fecha_vencimiento, estado, prioridad)
            VALUES (p_id_libro, p_id_usuario, NOW(),
                    DATE_ADD(CURDATE(), INTERVAL p_dias_vigencia DAY),
                    'PENDIENTE', 0);

            SET p_id_reserva = LAST_INSERT_ID();
            SET p_resultado = CONCAT('OK: Reserva creada. ID: ', p_id_reserva);
        END IF;
    END IF;
END$$

-- Mejor estructura para tabla auditoria
DROP TABLE IF EXISTS auditoria;
CREATE  TABLE auditoria (
                           id_auditoria INT NOT NULL AUTO_INCREMENT,
                           id_usuario INT NOT NULL COMMENT 'Usuario que realizó la acción',
                           tipo_accion ENUM('INSERT', 'UPDATE', 'DELETE', 'LOGIN', 'LOGOUT', 'CONSULTA', 'SISTEMA') NOT NULL,
                           tabla_afectada VARCHAR(50) COMMENT 'Tabla afectada (si aplica)',
                           id_registro_afectado INT COMMENT 'ID del registro afectado',

    -- Campos específicos para diferentes tipos de auditoría
                           modulo VARCHAR(50) COMMENT 'Módulo del sistema (ej: PRÉSTAMOS, USUARIOS)',
                           accion_especifica VARCHAR(100) NOT NULL COMMENT 'Descripción específica de la acción',

    -- Datos antes y después (para updates)
                           datos_anteriores JSON COMMENT 'Datos antes del cambio (solo UPDATE)',
                           datos_nuevos JSON COMMENT 'Datos después del cambio',

    -- Información de sesión
                           ip_address VARCHAR(45),
                           user_agent VARCHAR(255),
                           session_id VARCHAR(100),

    -- Metadatos
                           fecha_hora DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
                           exito BOOLEAN NOT NULL DEFAULT TRUE COMMENT 'Si la acción fue exitosa',
                           mensaje_error TEXT COMMENT 'Mensaje de error si falló',
                           tiempo_ejecucion_ms INT COMMENT 'Tiempo que tomó la acción',

                           PRIMARY KEY (id_auditoria),

    -- Índices para búsquedas rápidas
                           INDEX idx_usuario_fecha (id_usuario, fecha_hora),
                           INDEX idx_tabla_registro (tabla_afectada, id_registro_afectado),
                           INDEX idx_tipo_accion (tipo_accion),
                           INDEX idx_modulo (modulo),
                           INDEX idx_fecha (fecha_hora),
                           INDEX idx_exito (exito),

                           FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario)
                               ON DELETE RESTRICT
                               ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci
    COMMENT='Registro de auditoría de todas las acciones del sistema';

-- Vista para panel de administración
CREATE OR REPLACE VIEW v_auditoria_resumen AS
SELECT
    DATE(a.fecha_hora) as fecha,
    a.modulo,
    a.tipo_accion,
    COUNT(*) as total_acciones,
    COUNT(DISTINCT a.id_usuario) as usuarios_unicos,
    AVG(CASE WHEN a.exito = 1 THEN 1 ELSE 0 END) * 100 as porcentaje_exito
FROM auditoria a
GROUP BY DATE(a.fecha_hora), a.modulo, a.tipo_accion
ORDER BY fecha DESC, total_acciones DESC;

-- Vista para actividad sospechosa
CREATE OR REPLACE VIEW v_actividad_sospechosa AS
SELECT
    a.ip_address,
    u.nombre as usuario,
    COUNT(*) as acciones_5min,
    GROUP_CONCAT(DISTINCT a.modulo SEPARATOR ', ') as modulos_afectados,
    MIN(a.fecha_hora) as primera_accion,
    MAX(a.fecha_hora) as ultima_accion
FROM auditoria a
         LEFT JOIN usuario u ON a.id_usuario = u.id_usuario
WHERE a.fecha_hora >= DATE_SUB(NOW(), INTERVAL 5 MINUTE)
GROUP BY a.ip_address, u.id_usuario, u.nombre
HAVING COUNT(*) > 20  -- Más de 20 acciones en 5 minutos
ORDER BY acciones_5min DESC;
